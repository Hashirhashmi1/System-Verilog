// APB INTERFACE
interface apb_if(input logic pclk);
  logic presetn;
  logic [7:0] paddr;
  logic psel;
  logic penable;
  logic pwrite;
  logic [31:0] pwdata;
  logic [31:0] prdata;
  logic pready;
  logic pslverr;
endinterface

// APB DUT
module apb_dut(apb_if vif);
  logic [31:0] memory [0:255];

  assign vif.pready  = vif.psel && vif.penable;
  assign vif.prdata  = memory[vif.paddr];
  assign vif.pslverr = 1'b0;  

  always @(posedge vif.pclk) begin
    if(!vif.presetn) begin
      memory[0] <= 0;
      memory[1] <= 0;
      memory[2] <= 0;
      memory[3] <= 0;
      memory[4] <= 0;
      memory[5] <= 0;
      memory[6] <= 0;
      memory[7] <= 0;
      memory[8] <= 0;
    end
    else if(vif.psel && vif.penable && vif.pready && vif.pwrite) begin
      memory[vif.paddr] <= vif.pwdata;
      $display("[DUT] WRITE Address=%h Data=%h",vif.paddr,vif.pwdata);
    end
  end
endmodule
