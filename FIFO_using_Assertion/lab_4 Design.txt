`timescale 1ns/1ps
module fifo (
    input  logic       clk,
    input  logic       rst,
    input  logic       wr_en,
    input  logic       rd_en,
    input  logic [7:0] wr_data,
    output logic [7:0] rd_data,
    output logic       full,
    output logic       empty
);
    logic [7:0] mem [0:3];
    logic [2:0] count;
    logic [1:0] wr_ptr;
    logic [1:0] rd_ptr;

    assign full  = (count == 4);
    assign empty = (count == 0);

    always_ff @(posedge clk) begin
        if (rst) begin
            count   <= 0;
            wr_ptr  <= 0;
            rd_ptr  <= 0;
            rd_data <= 0;
        end
        else begin
            if (wr_en && !full) begin
                mem[wr_ptr] <= wr_data;
                wr_ptr      <= wr_ptr + 1;
                count       <= count + 1;
            end
            if (rd_en && !empty) begin
                rd_data <= mem[rd_ptr];
                rd_ptr  <= rd_ptr + 1;
                count   <= count - 1;
            end
        end
    end
endmodule
